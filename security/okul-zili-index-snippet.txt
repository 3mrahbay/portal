window.veliOkulZiliBildir = async function() {
  const ogr = (typeof veliAktifOgrenci !== "undefined" && veliAktifOgrenci) || (typeof veliOgrenciler !== "undefined" && veliOgrenciler[0]) || null;
  if (!ogr) return;
  const saat = document.getElementById("vzSaat")?.value || "";
  const kisi = document.getElementById("vzKisi")?.value || "Veli";
  const tarih = vzBugun();
  try {
    const tamVeri = {
      ogrenciId: ogr.id,
      ogrenciAd: ogr.ogrenciAdSoyad || "",
      sinif: (ayarListesi?.[ogr.id]?.kayit?.sinif) || ogr.sinif || "",
      tarih, hedefSaat: saat, alanKisi: kisi,
      durum: "yolda",
      veliEmail: (currentUser?.email || "").toLowerCase(),
      olusturuldu: new Date().toISOString(),
      guncellendi: serverTimestamp()
    };
    const batch = writeBatch(db);
    batch.set(doc(db, "pickupBildirimleri", ogr.id + "__" + tarih), tamVeri, { merge: true });
    const guvenliRef = doc(db, DANISMA_GUVENLI_KOLEKSIYONLAR.pickup, ogr.id + "__" + tarih);
    const guvenliMevcut = await getDoc(guvenliRef);
    if (guvenliMevcut.exists()) {
      batch.set(guvenliRef, {
        hedefSaat: saat, alanKisi: kisi, durum: "yolda",
        guncellendi: serverTimestamp()
      }, { merge: true });
    } else {
      batch.set(guvenliRef, okulZiliGuvenliVeri(tamVeri, ogr.id, tarih));
    }
    // İsimsiz kuyruk (ZEKY pickupKuyrukGuncelle ile aynı şema)
    batch.set(doc(db, "pickupKuyruk", tarih), {
      tarih, ogrenciler: { [ogr.id]: { saat } }, guncellendi: serverTimestamp()

async function personelSiniflariGetir(email, personelKaydi = null) {
  const lower = (email || "").trim().toLowerCase();
  if (!lower) return [];
  const belgeSiniflari = [
    ...(Array.isArray(personelKaydi?.siniflar) ? personelKaydi.siniflar : []),
    ...(Array.isArray(personelKaydi?.sinifAtamalari) ? personelKaydi.sinifAtamalari : [])
  ].map(s => String(s || "").trim()).filter(Boolean);
  if (belgeSiniflari.length) return [...new Set(belgeSiniflari)];

  // Eski personel kayıtlarında alanlar yoksa geriye dönük koleksiyon desteği.
  try {
    const q = query(collection(db, "sinifAtamalari"), where("email", "==", lower));
    const snap = await getDocs(q);
    const siniflar = [];
    snap.forEach(d => {
      const s = d.data().sinif;
      if (s && !siniflar.includes(s)) siniflar.push(s);
    });
    return siniflar;
  } catch (e) {
    console.warn("personelSiniflariGetir hatası:", e);
    return [];
  }
}
